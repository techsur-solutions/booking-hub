**scripts/traceability/analyze-coverage.sh:**

```bash
#!/bin/bash
set -euo pipefail

MATRIX=".planning/traceability-matrix.csv"
OUTPUT="docs/testing/test-gap-report.md"

echo "# Test Coverage Gap Analysis" > "$OUTPUT"
echo "" >> "$OUTPUT"
echo "**Generated:** $(date -u +%Y-%m-%dT%H:%M:%SZ)" >> "$OUTPUT"
echo "**Source:** Phase 8 traceability matrix analysis" >> "$OUTPUT"
echo "" >> "$OUTPUT"

# Overall statistics
TOTAL_FINDINGS=$(tail -n +2 "$MATRIX" | wc -l)
TESTED_FINDINGS=$(tail -n +2 "$MATRIX" | grep -E ',tested,|,verified,' | wc -l || echo 0)
COVERAGE_PCT=$((TESTED_FINDINGS * 100 / TOTAL_FINDINGS))

echo "## Summary" >> "$OUTPUT"
echo "" >> "$OUTPUT"
echo "- **Total findings:** $TOTAL_FINDINGS" >> "$OUTPUT"
echo "- **Tested findings:** $TESTED_FINDINGS" >> "$OUTPUT"
echo "- **Coverage:** ${COVERAGE_PCT}%" >> "$OUTPUT"
echo "" >> "$OUTPUT"

# Scan existing test files
echo "## Existing Test Inventory" >> "$OUTPUT"
echo "" >> "$OUTPUT"

BACKEND_TESTS=$(find services -name "*Test.java" 2>/dev/null | wc -l)
FRONTEND_TESTS=$(find frontend/e2e -name "*.spec.ts" 2>/dev/null | wc -l)

echo "- **Backend integration tests:** $BACKEND_TESTS files" >> "$OUTPUT"
echo "- **Frontend E2E tests:** $FRONTEND_TESTS files" >> "$OUTPUT"
echo "" >> "$OUTPUT"

# Per-feature gap analysis
echo "## Coverage by Feature" >> "$OUTPUT"
echo "" >> "$OUTPUT"

for feature in F1 F2 F3 F4 F5 F6 F7 F8 F9 F10 F11 F12 F13; do
  FEATURE_TOTAL=$(tail -n +2 "$MATRIX" | grep ",$feature" | wc -l || echo 0)
  if [ "$FEATURE_TOTAL" -eq 0 ]; then
    continue
  fi
  
  FEATURE_TESTED=$(tail -n +2 "$MATRIX" | grep ",$feature" | grep -E ',tested,|,verified,' | wc -l || echo 0)
  FEATURE_PCT=$((FEATURE_TESTED * 100 / FEATURE_TOTAL))
  
  # Get feature name from REQUIREMENTS.md
  FEATURE_NAME=$(grep -A 1 "^\*\*$feature\*\*:" .planning/REQUIREMENTS.md | tail -1 | sed 's/^[[:space:]]*-[[:space:]]*\[[[:space:]]*\][[:space:]]*//' || echo "Unknown")
  
  echo "### $feature: $FEATURE_NAME" >> "$OUTPUT"
  echo "" >> "$OUTPUT"
  echo "- **Coverage:** $FEATURE_TESTED/$FEATURE_TOTAL findings tested (${FEATURE_PCT}%)" >> "$OUTPUT"
  
  # List untested findings for this feature
  UNTESTED=$(tail -n +2 "$MATRIX" | grep ",$feature" | grep -v ',tested,' | grep -v ',verified,' | cut -d, -f1 || echo "")
  if [ -n "$UNTESTED" ]; then
    echo "- **Gaps:** $(echo "$UNTESTED" | tr '\n' ' ' | xargs)" >> "$OUTPUT"
  fi
  echo "" >> "$OUTPUT"
done

# High-priority gaps (explicit F13.3 edge cases)
echo "## High-Priority Gaps (F13.3 Edge Cases)" >> "$OUTPUT"
echo "" >> "$OUTPUT"
echo "Per FRD F13.3, these findings require explicit test coverage before release:" >> "$OUTPUT"
echo "" >> "$OUTPUT"

# Conflict detection edge cases (F2)
echo "### Conflict Detection (F2)" >> "$OUTPUT"
CONFLICT_GAPS=$(tail -n +2 "$MATRIX" | grep ',F2' | grep -v ',tested,' | grep -v ',verified,' | cut -d, -f1 || echo "")
if [ -n "$CONFLICT_GAPS" ]; then
  echo "- Findings: $(echo "$CONFLICT_GAPS" | tr '\n' ' ' | xargs)" >> "$OUTPUT"
  echo "- **Required tests:** Half-open interval semantics, location AND resource overlap, exclude booking's own prior state, exclude deleted/denied bookings" >> "$OUTPUT"
else
  echo "- ✓ All conflict detection findings tested" >> "$OUTPUT"
fi
echo "" >> "$OUTPUT"

# Approval/auto-approve interactions (F3)
echo "### Approval Workflow (F3)" >> "$OUTPUT"
APPROVAL_GAPS=$(tail -n +2 "$MATRIX" | grep ',F3' | grep -v ',tested,' | grep -v ',verified,' | cut -d, -f1 || echo "")
if [ -n "$APPROVAL_GAPS" ]; then
  echo "- Findings: $(echo "$APPROVAL_GAPS" | tr '\n' ' ' | xargs)" >> "$OUTPUT"
  echo "- **Required tests:** Auto-approve on/off transitions, one-way approve/deny (no re-toggling), permission-gated actions" >> "$OUTPUT"
else
  echo "- ✓ All approval workflow findings tested" >> "$OUTPUT"
fi
echo "" >> "$OUTPUT"

# Recurring booking scoping (F1.3)
echo "### Recurring Booking Scoping (F1.3)" >> "$OUTPUT"
RECURRING_GAPS=$(tail -n +2 "$MATRIX" | grep 'BF-011\|BF-015\|BF-017' | grep -v ',tested,' | grep -v ',verified,' | cut -d, -f1 || echo "")
if [ -n "$RECURRING_GAPS" ]; then
  echo "- Findings: $(echo "$RECURRING_GAPS" | tr '\n' ' ' | xargs)" >> "$OUTPUT"
  echo "- **Required tests:** Repeat-loop creates independent siblings, no series-aware edit/delete, repeatno upper bound" >> "$OUTPUT"
else
  echo "- ✓ All recurring booking findings tested" >> "$OUTPUT"
fi
echo "" >> "$OUTPUT"

# Permission matrix boundaries (F7)
echo "### Permission Matrix (F7)" >> "$OUTPUT"
PERMISSION_GAPS=$(tail -n +2 "$MATRIX" | grep ',F7' | grep -v ',tested,' | grep -v ',verified,' | cut -d, -f1 || echo "")
if [ -n "$PERMISSION_GAPS" ]; then
  echo "- Findings: $(echo "$PERMISSION_GAPS" | tr '\n' ' ' | xargs)" >> "$OUTPUT"
  echo "- **Required tests:** 17-flag inventory coverage, deny-by-default for unconfirmed flags, Tier1/Tier2 consistency" >> "$OUTPUT"
else
  echo "- ✓ All permission matrix findings tested" >> "$OUTPUT"
fi
echo "" >> "$OUTPUT"

# Open Questions needing tests
echo "## Open Questions Requiring Test Stubs" >> "$OUTPUT"
echo "" >> "$OUTPUT"
OPEN_QUESTIONS=$(tail -n +2 "$MATRIX" | grep 'Open Question' | cut -d, -f1 || echo "")
if [ -n "$OPEN_QUESTIONS" ]; then
  echo "Per F13.5, these Open Questions must have a test added when resolved:" >> "$OUTPUT"
  echo "" >> "$OUTPUT"
  echo "$(echo "$OPEN_QUESTIONS" | tr '\n' ' ' | xargs)" >> "$OUTPUT"
  echo "" >> "$OUTPUT"
  echo "See .planning/phases/01-legacy-functional-audit/open-questions.md for details." >> "$OUTPUT"
else
  echo "- ✓ No open questions pending resolution" >> "$OUTPUT"
fi
echo "" >> "$OUTPUT"

echo "---" >> "$OUTPUT"
echo "*Run \`bash scripts/traceability/analyze-coverage.sh\` after adding tests to update this report.*" >> "$OUTPUT"

echo "Coverage gap report written to $OUTPUT"
```

# Test Writing Guide: Legacy Parity Verification

**Audience:** Developers implementing features in Phases 4-7
**Purpose:** Write tests that verify legacy behavior equivalence per F0 audit findings

## Test Strategy by Layer

### Backend Integration Tests (Spring Boot)

**Pattern:** `@SpringBootTest` + `MockMvc` + WireMock for external services (Keycloak)

**File location:** `services/{service-name}/src/test/java/.../controller/*IntegrationTest.java`

**Template:**
```java
@SpringBootTest(webEnvironment = SpringBootTest.WebEnvironment.RANDOM_PORT)
@ActiveProfiles("test")
@AutoConfigureMockMvc
class MyFeatureIntegrationTest {
    
    @org.junit.jupiter.api.extension.RegisterExtension
    static WireMockExtension wireMock = WireMockExtension.newInstance()
            .options(wireMockConfig().dynamicPort())
            .build();

    @DynamicPropertySource
    static void configureProperties(DynamicPropertyRegistry registry) {
        registry.add("keycloak.server-url", () -> wireMock.baseUrl());
    }

    @Autowired
    private MockMvc mockMvc;

    @Autowired
    private MyRepository repository;

    @BeforeEach
    void setup() {
        repository.deleteAll();
        // Stub Keycloak responses
        wireMock.stubFor(post(urlPathMatching("/realms/.*/protocol/openid-connect/token"))
            .willReturn(aResponse().withStatus(200).withBody("{\"access_token\":\"test\"}")));
    }

    @Test
    void testFeatureBehavior() throws Exception {
        // GIVEN: setup test data
        // WHEN: perform request with JWT mock
        mockMvc.perform(post("/api/resource")
                .with(jwt().authorities(new SimpleGrantedAuthority("SCOPE_admin")))
                .contentType(MediaType.APPLICATION_JSON)
                .content("{\"field\":\"value\"}"))
            // THEN: assert response and side effects
            .andExpect(status().isCreated())
            .andExpect(jsonPath("$.id").exists());
        
        // Assert DB state
        assertThat(repository.findAll()).hasSize(1);
    }
}
```

**Key practices:**
- Use WireMock for Keycloak Admin API stubs (see `UserControllerIntegrationTest` example)
- Test against real Postgres from docker-compose (not Testcontainers — Docker API compatibility issue)
- Use `.with(jwt().authorities(...))` for authentication
- Verify both response contract AND database state
- Test happy path + one failure case per endpoint

### Frontend E2E Tests (Playwright)

**Pattern:** Full-browser testing with VITE_E2E_MODE bypass for Keycloak

**File location:** `frontend/e2e/{feature}.spec.ts`

**Template:**
```typescript
import { test, expect } from '@playwright/test';

test.describe('Feature Name', () => {
  test.beforeEach(async ({ page }) => {
    // E2E mode bypasses Keycloak auth
    await page.goto('/feature-path');
  });

  test('user can perform action', async ({ page }) => {
    // GIVEN: navigate to feature page
    await expect(page.locator('h1')).toHaveText('Feature Page');
    
    // WHEN: interact with UI
    await page.fill('input[name="field"]', 'value');
    await page.click('button[type="submit"]');
    
    // THEN: verify outcome
    await expect(page.locator('.success-message')).toBeVisible();
  });
});
```

**Key practices:**
- Tests run with `VITE_E2E_MODE=true` (bypasses Keycloak, see `playwright.config.ts`)
- Test navigation flow, form submission, data rendering
- Use accessibility-friendly selectors (role/label/text, not brittle classes)
- Keep tests focused (one user story per test file)

## High-Priority Edge Cases (F13.3)

### Conflict Detection (F2)

**Test scenarios:**
1. **Half-open interval semantics:** Booking A ends at 10:00, Booking B starts at 10:00 → NO conflict
2. **Location overlap:** Same location, overlapping times → conflict detected
3. **Resource overlap:** Different locations, shared resource, overlapping times → conflict detected
4. **Exclude own prior state:** Editing Booking A to new time that overlaps A's old time → NO self-conflict
5. **Exclude deleted/denied:** Active booking overlaps a denied booking → NO conflict (denied doesn't block)

**Test class:** `ConflictDetectionEngineTest.java`
```java
@Test
void halfOpenInterval_noConflict() {
    // Event A: 09:00-10:00
    // Event B: 10:00-11:00
    // Expected: no conflict (10:00 start == 10:00 end is boundary, not overlap)
    boolean conflict = conflictEngine.detectConflict(locationId, startB, endB, excludeBookingId);
    assertThat(conflict).isFalse();
}
```

### Approval Workflow (F3)

**Test scenarios:**
1. **Auto-approve off:** New booking → `pending` status
2. **Auto-approve on:** New booking → `approved` status
3. **One-way approve:** `pending` → `approved` is allowed, `approved` → `pending` is rejected
4. **Permission gate:** User without `allowApproveBooking` cannot call `/api/bookings/{id}/approve`

### Recurring Booking Scoping (F1.3)

**Test scenarios:**
1. **Independent siblings:** Creating 3-occurrence recurring booking produces 3 independent DB rows with no shared series ID
2. **Single-occurrence delete:** Deleting one occurrence leaves others intact
3. **No series-aware edit:** Editing one occurrence does not propagate to siblings

### Permission Matrix (F7)

**Test scenarios:**
1. **17-flag coverage:** All 17 permission flags from BF-080 have at least one gated action tested
2. **Deny-by-default:** Undefined flag returns 403 (not 500, not allow)
3. **Tier1/Tier2 consistency:** Gateway and backend service agree on permission requirements (see `Tier1Tier2ConsistencyTest`)
