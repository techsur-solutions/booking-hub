  <action>
Create useViewState hook wrapping React Router's useSearchParams for reading/writing view state to URL:

```typescript
import { useSearchParams } from 'react-router-dom';

export interface ViewState {
  date?: string;           // ISO date YYYY-MM-DD
  view?: 'calendar' | 'day' | 'list';
  locationId?: string;     // UUID
  status?: string;         // pending|approved|denied
  query?: string;          // search query
  startDate?: string;      // for List View date range
  endDate?: string;        // for List View date range
}

export function useViewState() {
  const [searchParams, setSearchParams] = useSearchParams();

  const getState = (): ViewState => ({
    date: searchParams.get('date') || new Date().toISOString().split('T')[0],
    view: (searchParams.get('view') as ViewState['view']) || 'calendar',
    locationId: searchParams.get('location_id') || undefined,
    status: searchParams.get('status') || undefined,
    query: searchParams.get('q') || undefined,
    startDate: searchParams.get('start_date') || undefined,
    endDate: searchParams.get('end_date') || undefined,
  });

  const updateState = (updates: Partial<ViewState>) => {
    const current = Object.fromEntries(searchParams.entries());
    const merged = { ...current, ...updates };
    // Remove undefined values
    Object.keys(merged).forEach(key => {
      if (merged[key] === undefined) delete merged[key];
    });
    setSearchParams(merged);
  };

  return { ...getState(), updateState };
}
```

Hook merges updates into existing URL params (doesn't replace entire query string).
