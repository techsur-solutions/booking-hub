**Button.tsx** — Reusable button with variants:
```typescript
export interface ButtonProps extends React.ButtonHTMLAttributes<HTMLButtonElement> {
  variant?: 'primary' | 'secondary' | 'danger' | 'ghost';
  size?: 'small' | 'medium' | 'large';
  isLoading?: boolean;
}

export function Button({ variant = 'primary', size = 'medium', isLoading, children, ...props }: ButtonProps) {
  return (
    <button
      className={`btn btn--${variant} btn--${size}`}
      disabled={isLoading || props.disabled}
      {...props}
    >
      {isLoading ? <span className="btn__spinner" /> : children}
    </button>
  );
}
```

**Input.tsx** — Text input with validation error:
```typescript
export interface InputProps extends React.InputHTMLAttributes<HTMLInputElement> {
  label?: string;
  error?: string;
}

export function Input({ label, error, ...props }: InputProps) {
  return (
    <div className="form-field">
      {label && <label htmlFor={props.id}>{label}</label>}
      <input
        className={`form-input ${error ? 'form-input--error' : ''}`}
        aria-invalid={!!error}
        aria-describedby={error ? `${props.id}-error` : undefined}
        {...props}
      />
      {error && <span id={`${props.id}-error`} className="form-error">{error}</span>}
    </div>
  );
}
```

**Select.tsx**, **Textarea.tsx** — Follow same pattern with label/error props.

**Form.tsx** — Form wrapper with validation error summary:
```typescript
export interface FormProps extends React.FormHTMLAttributes<HTMLFormElement> {
  errors?: Record<string, string>;
}

export function Form({ errors, children, ...props }: FormProps) {
  const errorCount = errors ? Object.keys(errors).length : 0;
  return (
    <form {...props}>
      {errorCount > 0 && (
        <div role="alert" className="form-errors">
          <strong>{errorCount} error{errorCount > 1 ? 's' : ''} found:</strong>
          <ul>
            {Object.entries(errors!).map(([field, message]) => (
              <li key={field}>{message}</li>
            ))}
          </ul>
        </div>
      )}
      {children}
    </form>
  );
}
```

**Modal.tsx** — Full-featured modal with portal rendering:
```typescript
import { createPortal } from 'react-dom';

export interface ModalProps {
  isOpen: boolean;
  onClose: () => void;
  title?: string;
  children: ReactNode;
  footer?: ReactNode;
}

export function Modal({ isOpen, onClose, title, children, footer }: ModalProps) {
  if (!isOpen) return null;

  return createPortal(
    <div className="modal-backdrop" onClick={onClose}>
      <div className="modal" onClick={(e) => e.stopPropagation()} role="dialog" aria-modal="true">
        {title && (
          <div className="modal__header">
            <h2>{title}</h2>
            <button onClick={onClose} aria-label="Close">&times;</button>
          </div>
        )}
        <div className="modal__body">{children}</div>
        {footer && <div className="modal__footer">{footer}</div>}
      </div>
    </div>,
    document.body
  );
}
```
