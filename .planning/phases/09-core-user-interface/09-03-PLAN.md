**StateConfirmationBanner.tsx** — Reusable banner for explicit state confirmation per UI-PATTERN-01:

```typescript
import React, { useEffect } from 'react';

export interface StateConfirmationBannerProps {
  action: string; // "Booking saved", "Approved", "Denied", "Settings updated"
  entity?: string; // "Booking #482", "Location 'Studio 1'", optional
  timestamp: Date;
  autoDismiss?: number; // milliseconds, default 5000
  onDismiss: () => void;
  variant?: 'success' | 'warning' | 'error'; // styling variant
}

export function StateConfirmationBanner({
  action,
  entity,
  timestamp,
  autoDismiss = 5000,
  onDismiss,
  variant = 'success',
}: StateConfirmationBannerProps) {
  useEffect(() => {
    if (autoDismiss > 0) {
      const timer = setTimeout(onDismiss, autoDismiss);
      return () => clearTimeout(timer);
    }
  }, [autoDismiss, onDismiss]);

  const timeStr = timestamp.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });

  return (
    <div
      role="alert"
      aria-live="polite"
      className={`confirmation-banner confirmation-banner--${variant}`}
      data-testid="state-confirmation-banner"
    >
      <span className="confirmation-banner__icon" aria-hidden="true">
        {variant === 'success' ? '✓' : variant === 'error' ? '✕' : '⚠'}
      </span>
      <div className="confirmation-banner__content">
        <strong>{action}</strong>
        {entity && <span> — {entity}</span>}
        <time className="confirmation-banner__timestamp" dateTime={timestamp.toISOString()}>
          {timeStr}
        </time>
      </div>
      <button
        type="button"
        onClick={onDismiss}
        className="confirmation-banner__dismiss"
        aria-label="Dismiss"
      >
        ✕
      </button>
    </div>
  );
}
```

**ConflictBadge.tsx** — Single conflict badge component per UI-PATTERN-02:

```typescript
import React from 'react';
import { ConflictFlag } from '../api/types';

export interface ConflictBadgeProps {
  conflicts: ConflictFlag[];
  onClick?: (bookingId: string) => void; // Optional click handler to open detail modal
  size?: 'small' | 'medium' | 'large'; // Size variant for different contexts
}

export function ConflictBadge({ conflicts, onClick, size = 'medium' }: ConflictBadgeProps) {
  if (!conflicts || conflicts.length === 0) return null;

  const handleClick = () => {
    if (onClick && conflicts.length > 0) {
      // Open first conflicting booking's detail modal
      onClick(conflicts[0].conflicting_booking_id);
    }
  };

  const tooltipText = conflicts.map(c => 
    `Conflicts with Booking #${c.conflicting_booking_id} (${c.conflict_type})`
  ).join('; ');

  return (
    <span
      className={`conflict-badge conflict-badge--${size}`}
      title={tooltipText}
      onClick={handleClick}
      onKeyDown={(e) => e.key === 'Enter' && handleClick()}
      role={onClick ? 'button' : undefined}
      tabIndex={onClick ? 0 : undefined}
      aria-label={`Conflict warning: ${tooltipText}`}
      data-testid="conflict-badge"
    >
      <span className="conflict-badge__icon" aria-hidden="true">⚠</span>
      <span className="conflict-badge__count">{conflicts.length}</span>
    </span>
  );
}
```

Styling: red ⚠ icon, dashed red outline when applied to booking blocks, consistent across Calendar/Day/List/Queue/Detail views.

**permissions.ts** — Helper functions for role checks:

```typescript
import keycloak from '../auth/keycloak';

export function hasRole(roleName: string): boolean {
  return keycloak.realmAccess?.roles?.includes(roleName) ?? false;
}

export function hasAnyRole(roleNames: string[]): boolean {
  return roleNames.some(role => hasRole(role));
}

export function hasAllRoles(roleNames: string[]): boolean {
  return roleNames.every(role => hasRole(role));
}

export function hasApprovePermission(): boolean {
  return hasRole('role_booking_approver');
}

export function hasAnyAdminPermission(): boolean {
  return hasAnyRole([
    'role_location_manager',
    'role_permission_manager',
    'role_audit_viewer',
  ]);
}
```

**PermissionGate.tsx** — HOC for permission-gated rendering per UI-PATTERN-03:

```typescript
import React, { ReactNode } from 'react';
import { hasRole, hasAnyRole } from '../utils/permissions';

export interface PermissionGateProps {
  children: ReactNode;
  requireRole?: string; // Single required role
  requireAnyRole?: string[]; // At least one of these roles
  fallback?: ReactNode; // Optional fallback when permission denied (default: null)
}

export function PermissionGate({
  children,
  requireRole,
  requireAnyRole,
  fallback = null,
}: PermissionGateProps) {
  let hasPermission = true;

  if (requireRole) {
    hasPermission = hasRole(requireRole);
  } else if (requireAnyRole) {
    hasPermission = hasAnyRole(requireAnyRole);
  }

  if (!hasPermission) {
    return <>{fallback}</>;
  }

  return <>{children}</>;
}
```

Usage example:
```tsx
<PermissionGate requireRole="role_booking_approver">
  <Link to="/approvals">Approvals</Link>
</PermissionGate>
```
