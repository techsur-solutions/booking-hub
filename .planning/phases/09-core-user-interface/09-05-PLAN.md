  <action>
Install `@fullcalendar/react@^6.1.0`, `@fullcalendar/daygrid`, `@fullcalendar/timegrid`, `@fullcalendar/interaction`.

Create FullCalendarWrapper wrapping FullCalendar with booking data from useBookings hook:

```tsx
import FullCalendar from '@fullcalendar/react';
import dayGridPlugin from '@fullcalendar/daygrid';
import timeGridPlugin from '@fullcalendar/timegrid';
import interactionPlugin from '@fullcalendar/interaction';
import { useBookings } from '../../hooks/useBookings';
import { useLocations } from '../../hooks/useLocations';

export function FullCalendarWrapper() {
  const { data: bookings, isLoading } = useBookings({});
  const { data: locations } = useLocations();
  
  const events = bookings?.map(booking => {
    const location = locations?.find(l => l.id === booking.location_id);
    return {
      id: booking.id,
      title: booking.title,
      start: booking.start_time,
      end: booking.end_time,
      backgroundColor: location?.colour || '#3b82f6',
      extendedProps: { booking, hasConflict: booking.conflict_flags.length > 0 },
    };
  }) || [];

  return (
    <FullCalendar
      plugins={[dayGridPlugin, timeGridPlugin, interactionPlugin]}
      initialView="dayGridMonth"
      headerToolbar={{ left: 'prev,next today', center: 'title', right: 'dayGridMonth,timeGridWeek' }}
      events={events}
      dateClick={handleDateClick}
      eventClick={handleEventClick}
      eventContent={renderEventContent}
    />
  );
}
```

handleDateClick opens Booking Create Form pre-filled. handleEventClick opens Booking Detail Modal.
