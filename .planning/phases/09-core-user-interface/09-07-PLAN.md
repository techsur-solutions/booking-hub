```typescript
export function exportToCSV(bookings: Booking[]): void {
  const headers = ['Title', 'Location', 'Resources', 'Start Time', 'End Time', 'Status', 'Has Conflicts'];
  const rows = bookings.map(b => [
    b.title,
    b.location?.name || '',
    b.resources?.map(r => r.name).join('; ') || '',
    b.start_time,
    b.end_time,
    b.status,
    b.conflict_flags.length > 0 ? 'Yes' : 'No'
  ]);
  const csv = [headers, ...rows].map(row => row.join(',')).join('\n');
  downloadFile(csv, 'bookings.csv', 'text/csv');
}

export function exportToJSON(bookings: Booking[]): void {
  const json = JSON.stringify(bookings, null, 2);
  downloadFile(json, 'bookings.json', 'application/json');
}

function downloadFile(content: string, filename: string, mimeType: string): void {
  const blob = new Blob([content], { type: mimeType });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = filename;
  a.click();
  URL.revokeObjectURL(url);
}
```

Export buttons in ListView trigger these functions with currently filtered bookings (not all bookings, only visible ones).
